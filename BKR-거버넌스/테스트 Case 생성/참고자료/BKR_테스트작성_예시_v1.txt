[BDDMockito 스타일 예시]
@Test
@DisplayName("정상 고객 조회 - 유효한 ID 입력 시 고객 정보 반환")
void should_return_customer_when_findById_called_with_valid_id() {
    // Given
    given(customerRepository.findById(1L)).willReturn(Optional.of(mockCustomer));

    // When
    CustomerDto result = customerService.findById(1L);

    // Then
    assertThat(result).isNotNull();
    assertThat(result.getId()).isEqualTo(1L);
}

[예외 케이스 예시]
@Test
@DisplayName("존재하지 않는 고객 조회 - NotFoundException 발생")
void should_throw_not_found_when_findById_with_nonexistent_id() {
    // Given
    given(customerRepository.findById(999L)).willReturn(Optional.empty());

    // When & Then
    assertThatThrownBy(() -> customerService.findById(999L))
        .isInstanceOf(BkrNotFoundException.class)
        .hasMessageContaining("고객을 찾을 수 없습니다");
}
